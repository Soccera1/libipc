cmake_minimum_required(VERSION 3.16)
project(libipc_consumer C)
option(USE_STATIC "Test static exported target" OFF)
option(USE_PKG_CONFIG "Test installed pkg-config metadata" OFF)
add_executable(consumer main.c)
if(USE_PKG_CONFIG)
  find_package(PkgConfig REQUIRED)
  if(USE_STATIC)
    pkg_check_modules(IPC REQUIRED IMPORTED_TARGET libipc-static>=0.1)
  else()
    pkg_check_modules(IPC REQUIRED IMPORTED_TARGET libipc>=0.1)
  endif()
  target_link_libraries(consumer PRIVATE PkgConfig::IPC)
else()
  find_package(libipc 0.1 CONFIG REQUIRED)
  if(USE_STATIC)
    target_link_libraries(consumer PRIVATE libipc::static)
  else()
    target_link_libraries(consumer PRIVATE libipc::shared)
  endif()
endif()
