cmake_minimum_required(VERSION 3.16)
project(libipc VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
option(IPC_BUILD_CPP_TEST "Build the C++ ownership smoke test" OFF)
option(IPC_BUILD_EXAMPLES "Build portable example applications" ON)
option(IPC_BUILD_BENCHMARKS "Build round-trip latency and throughput benchmark" ON)
if(WIN32)
  set(default_message win32_pipe)
  set(default_shm win32_mapping)
  set(default_reactor iocp)
else()
  set(default_message unix)
  set(default_shm posix)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(default_reactor epoll)
  elseif(CMAKE_SYSTEM_NAME MATCHES "Darwin|FreeBSD|OpenBSD|NetBSD|DragonFly")
    set(default_reactor kqueue)
  else()
    set(default_reactor poll)
  endif()
endif()
set(IPC_MESSAGE_BACKEND ${default_message} CACHE STRING "Messaging backend")
set(IPC_SHM_BACKEND ${default_shm} CACHE STRING "Shared memory backend")
set(IPC_EVENT_BACKEND ${default_reactor} CACHE STRING "Event reactor")
set_property(CACHE IPC_MESSAGE_BACKEND PROPERTY STRINGS unix win32_pipe)
set_property(CACHE IPC_SHM_BACKEND PROPERTY STRINGS posix win32_mapping)
set_property(CACHE IPC_EVENT_BACKEND PROPERTY STRINGS poll epoll kqueue iocp)
set(THREADS_PREFER_PTHREAD_FLAG TRUE)
find_package(Threads REQUIRED)
include(CheckLibraryExists)
include(CheckSymbolExists)
include(CheckCSourceCompiles)
set(IPC_SYSTEM_LIBRARIES Threads::Threads)
if(WIN32)
  if(NOT IPC_MESSAGE_BACKEND STREQUAL "win32_pipe" OR NOT IPC_SHM_BACKEND STREQUAL "win32_mapping" OR NOT IPC_EVENT_BACKEND STREQUAL "iocp")
    message(FATAL_ERROR "Windows requires win32_pipe/win32_mapping/iocp")
  endif()
  set(IPC_SOURCES src/ipc_win32.c)
  list(APPEND IPC_SYSTEM_LIBRARIES advapi32)
elseif(UNIX)
  if(NOT IPC_MESSAGE_BACKEND STREQUAL "unix" OR NOT IPC_SHM_BACKEND STREQUAL "posix")
    message(FATAL_ERROR "Unix requires unix/posix backends")
  endif()
  set(IPC_FEATURE_DEFINITIONS _POSIX_C_SOURCE=200112L _REENTRANT)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(IPC_FEATURE_DEFINITIONS _POSIX_C_SOURCE=200809L _GNU_SOURCE _REENTRANT)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "DragonFly")
    # Its kqueue header needs BSD typedefs; the default namespace also exposes poll.
    set(IPC_FEATURE_DEFINITIONS _REENTRANT)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    list(APPEND IPC_FEATURE_DEFINITIONS _DARWIN_C_SOURCE)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "AIX")
    list(APPEND IPC_FEATURE_DEFINITIONS _ALL_SOURCE _THREAD_SAFE)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "HP-UX")
    list(APPEND IPC_FEATURE_DEFINITIONS _HPUX_SOURCE)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "SunOS")
    list(APPEND IPC_FEATURE_DEFINITIONS __EXTENSIONS__)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
    list(APPEND IPC_FEATURE_DEFINITIONS _NETBSD_SOURCE)
  endif()
  foreach(definition IN LISTS IPC_FEATURE_DEFINITIONS)
    list(APPEND CMAKE_REQUIRED_DEFINITIONS -D${definition})
  endforeach()
  check_library_exists(rt shm_open "" IPC_NEEDS_RT)
  if(IPC_NEEDS_RT)
    list(APPEND IPC_SYSTEM_LIBRARIES rt)
  endif()
  foreach(lib socket nsl)
    if(CMAKE_SYSTEM_NAME STREQUAL "SunOS")
      list(APPEND IPC_SYSTEM_LIBRARIES ${lib})
    endif()
  endforeach()
  set(CMAKE_REQUIRED_LIBRARIES ${IPC_SYSTEM_LIBRARIES})
  foreach(symbol shm_open sem_open clock_gettime poll)
    check_symbol_exists(${symbol} "sys/types.h;sys/mman.h;semaphore.h;time.h;poll.h" IPC_HAVE_${symbol})
    if(NOT IPC_HAVE_${symbol})
      message(FATAL_ERROR "Required POSIX interface ${symbol} is unavailable")
    endif()
  endforeach()
  check_symbol_exists(sem_clockwait "semaphore.h;time.h" IPC_HAVE_SEM_CLOCKWAIT)
  set(IPC_SOURCES src/ipc.c)
  if(IPC_EVENT_BACKEND STREQUAL "poll")
    list(APPEND IPC_SOURCES src/reactor_poll.c)
  elseif(IPC_EVENT_BACKEND STREQUAL "epoll")
    check_symbol_exists(epoll_create1 "sys/epoll.h" IPC_HAVE_EPOLL)
    if(NOT IPC_HAVE_EPOLL)
      message(FATAL_ERROR "Selected epoll reactor is unavailable")
    endif()
    set(IPC_REACTOR_EPOLL 1)
    list(APPEND IPC_SOURCES src/reactor_native.c)
  elseif(IPC_EVENT_BACKEND STREQUAL "kqueue")
    check_symbol_exists(kqueue "sys/types.h;sys/event.h" IPC_HAVE_KQUEUE)
    if(NOT IPC_HAVE_KQUEUE)
      message(FATAL_ERROR "Selected kqueue reactor is unavailable")
    endif()
    list(APPEND IPC_SOURCES src/reactor_native.c)
  else()
    message(FATAL_ERROR "Unsupported Unix reactor: ${IPC_EVENT_BACKEND}")
  endif()
else()
  message(FATAL_ERROR "Unsupported platform; supply a Unix or Windows toolchain")
endif()
configure_file(cmake/ipc_config.h.in ipc_config.h)
set(IPC_RUST_LIBRARIES "")
foreach(lib IN LISTS IPC_SYSTEM_LIBRARIES)
  if(NOT lib STREQUAL "Threads::Threads")
    string(APPEND IPC_RUST_LIBRARIES "${lib}\n")
  endif()
endforeach()
foreach(flag IN LISTS CMAKE_THREAD_LIBS_INIT)
  if(flag STREQUAL "-pthread")
    string(APPEND IPC_RUST_LIBRARIES "pthread\n")
  elseif(flag MATCHES "^-l(.+)$")
    string(APPEND IPC_RUST_LIBRARIES "${CMAKE_MATCH_1}\n")
  endif()
endforeach()
if(WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "AIX")
  set(IPC_STATIC_NAME ipc_static)
else()
  set(IPC_STATIC_NAME ipc)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ipc-rust-link.txt" CONTENT "${IPC_STATIC_NAME}\n${IPC_RUST_LIBRARIES}")
set(IPC_PC_SYSTEM_LIBRARIES "${CMAKE_THREAD_LIBS_INIT}")
foreach(lib IN LISTS IPC_SYSTEM_LIBRARIES)
  if(NOT lib STREQUAL "Threads::Threads")
    string(APPEND IPC_PC_SYSTEM_LIBRARIES " -l${lib}")
  endif()
endforeach()
string(STRIP "${IPC_PC_SYSTEM_LIBRARIES}" IPC_PC_SYSTEM_LIBRARIES)
file(RELATIVE_PATH IPC_PC_PREFIX "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
configure_file(cmake/libipc.pc.in libipc.pc @ONLY)
configure_file(cmake/libipc-static.pc.in libipc-static.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libipc.pc ${CMAKE_CURRENT_BINARY_DIR}/libipc-static.pc
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
add_library(ipc_objects OBJECT ${IPC_SOURCES})
set_target_properties(ipc_objects PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 99 C_STANDARD_REQUIRED ON)
target_include_directories(ipc_objects PRIVATE include ${CMAKE_CURRENT_BINARY_DIR})
if(UNIX)
  target_compile_definitions(ipc_objects PRIVATE ${IPC_FEATURE_DEFINITIONS})
else()
  target_compile_definitions(ipc_objects PRIVATE _WIN32_WINNT=0x0601)
endif()
target_link_libraries(ipc_objects PRIVATE ${IPC_SYSTEM_LIBRARIES})
foreach(kind shared static)
  string(TOUPPER ${kind} type)
  add_library(ipc_${kind} ${type} $<TARGET_OBJECTS:ipc_objects>)
  add_library(libipc::${kind} ALIAS ipc_${kind})
  set_target_properties(ipc_${kind} PROPERTIES OUTPUT_NAME ipc EXPORT_NAME ${kind})
  target_include_directories(ipc_${kind} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(ipc_${kind} PUBLIC ${IPC_SYSTEM_LIBRARIES})
endforeach()
set_target_properties(ipc_shared PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0 WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
  # Its library search uses libNAME.so.MAJOR.MINOR, without a patch component.
  set(IPC_OPENBSD_ABI "${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}")
  set_target_properties(ipc_shared PROPERTIES VERSION ${IPC_OPENBSD_ABI} SOVERSION ${IPC_OPENBSD_ABI})
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "AIX" OR WIN32)
  set_target_properties(ipc_static PROPERTIES OUTPUT_NAME ipc_static)
endif()
install(TARGETS ipc_shared ipc_static EXPORT libipcTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
# A distinct archive name lets pkg-config select static linkage without
# nonportable linker switches. Keep the conventional libipc.a installation too.
if(IPC_STATIC_NAME STREQUAL "ipc")
  install(FILES $<TARGET_FILE:ipc_static> DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RENAME "${CMAKE_STATIC_LIBRARY_PREFIX}ipc_static${CMAKE_STATIC_LIBRARY_SUFFIX}")
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT libipcTargets NAMESPACE libipc:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libipc)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/libipcConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/libipcConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libipc)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/libipcConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libipcConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/libipcConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libipc)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/libipc)
if(IPC_BUILD_EXAMPLES)
  add_executable(ipc_echo examples/echo.c)
  add_executable(ipc_shm examples/shared_memory.c)
  target_link_libraries(ipc_echo PRIVATE ipc_static)
  target_link_libraries(ipc_shm PRIVATE ipc_static)
endif()
if(IPC_BUILD_BENCHMARKS)
  add_executable(ipc_benchmark benchmarks/roundtrip.c)
  target_link_libraries(ipc_benchmark PRIVATE ipc_static)
endif()
if(BUILD_TESTING)
  add_test(NAME ipc_reject_invalid_configuration COMMAND ${CMAKE_COMMAND}
    "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}" "-DBINARY=${CMAKE_CURRENT_BINARY_DIR}/rejection"
    "-DTOOLCHAIN=${CMAKE_TOOLCHAIN_FILE}" "-DCOMPILER=${CMAKE_C_COMPILER}"
    "-DC_FLAGS=${CMAKE_C_FLAGS}" "-DGENERATOR=${CMAKE_GENERATOR}"
    "-DPLATFORM=${CMAKE_GENERATOR_PLATFORM}" "-DTOOLSET=${CMAKE_GENERATOR_TOOLSET}"
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/configure.cmake)
  if(IPC_BUILD_CPP_TEST)
    enable_language(CXX)
    add_executable(ipc_cpp_test tests/cpp.cpp tests/process.c)
    set_target_properties(ipc_cpp_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
    target_link_libraries(ipc_cpp_test PRIVATE ipc_static)
    add_test(NAME ipc_cpp_ownership COMMAND ipc_cpp_test)
    if(TARGET ipc_echo)
      add_test(NAME ipc_cpp_c_interop COMMAND ipc_cpp_test $<TARGET_FILE:ipc_echo>)
      set_tests_properties(ipc_cpp_c_interop PROPERTIES TIMEOUT 15)
    endif()
  endif()
  if(WIN32)
    add_executable(ipc_test tests/test_win32.c)
    add_executable(ipc_crash_test tests/crash_win32.c)
  else()
    add_executable(ipc_test tests/test.c)
    add_executable(ipc_crash_test tests/crash.c)
  endif()
  add_executable(ipc_framing_test tests/framing.c)
  target_link_libraries(ipc_framing_test PRIVATE ipc_static)
  add_test(NAME ipc_adversarial_framing COMMAND ipc_framing_test)
  set_tests_properties(ipc_adversarial_framing PROPERTIES TIMEOUT 15)
  target_link_libraries(ipc_test PRIVATE ipc_static)
  target_link_libraries(ipc_crash_test PRIVATE ipc_static)
  add_test(NAME ipc_crash_cleanup COMMAND ipc_crash_test)
  set_tests_properties(ipc_crash_cleanup PROPERTIES TIMEOUT 15)
  add_test(NAME ipc_integration COMMAND ipc_test)
  set_tests_properties(ipc_integration PROPERTIES TIMEOUT 15)
  foreach(kind shared static)
    add_executable(ipc_behavior_${kind} tests/behavior.c)
    target_link_libraries(ipc_behavior_${kind} PRIVATE ipc_${kind})
    add_test(NAME ipc_behavior_${kind} COMMAND ipc_behavior_${kind})
    set_tests_properties(ipc_behavior_${kind} PROPERTIES TIMEOUT 20)
  endforeach()
  foreach(test_target ipc_test ipc_cpp_test)
    if(TARGET ${test_target})
      if(MSVC)
        target_compile_options(${test_target} PRIVATE /UNDEBUG)
      else()
        target_compile_options(${test_target} PRIVATE -UNDEBUG)
      endif()
    endif()
  endforeach()
endif()
